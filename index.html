Backend Script (Google Apps Script)

Salin kode ini dan pasang pada Google Sheets target untuk menerima sinkronisasi data.

// =====================================================================
// KONFIGURASI BACKEND SCRIPT PRESENSI SCHOOL PRIME
// Spreadsheet ID Target: 1a7IIwqvRID3LdWi67XuEVTBqgSpveGlktiKV4DJTPpY
// Nama Sheet Target: Data Absensi
// =====================================================================

var SPREADSHEET_ID = "1a7IIwqvRID3LdWi67XuEVTBqgSpveGlktiKV4DJTPpY";
var SHEET_NAME = "Data Absensi";

function getSheet() {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(["NIS", "Nama Lengkap", "Kelas", "Tanggal Absen", "Keterangan", "kode RFID", "Waktu", "Jenis Absen"]);
  }
  return sheet;
}

// Handler POST: Menyimpan data presensi baru
function doPost(e) {
  try {
    var sheet = getSheet();
    var data = JSON.parse(e.postData.contents);
    
    if (Array.isArray(data)) {
      data.forEach(function(row) {
        sheet.appendRow([
          row.nis || "",
          row.nama || "",
          row.kelas || "",
          row.tanggal || "",
          row.keterangan || "",
          row.kodeRfid || "",
          row.waktu || "",
          row.jenisAbsen || "Absen Masuk"
        ]);
      });
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: data.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Handler GET: Membaca database untuk pemeriksaan duplikasi
function doGet(e) {
  try {
    var sheet = getSheet();
    var rows = sheet.getDataRange().getValues();
    var result = [];
    if (rows.length > 1) {
      for (var i = 1; i < rows.length; i++) {
        result.push({
          nis: rows[i][0],
          nama: rows[i][1],
          kelas: rows[i][2],
          tanggal: rows[i][3],
          keterangan: rows[i][4],
          kodeRfid: rows[i][5],
          waktu: rows[i][6],
          jenisAbsen: rows[i][7]
        });
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ status: "success", data: result }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
