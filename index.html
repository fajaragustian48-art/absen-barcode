Panduan Setup Backend Apps Script

Pasang kode Google Apps Script di Spreadsheet Anda agar sistem dapat menyinkronkan data secara otomatis ke sheet Data Absensi (GID: 1-S1Xwha8cuZvNXpkXpU_3m9dam7Nu9cPU6vUYyXN5Dk).

1. Buka Apps Script

Buka Spreadsheet Google Sheets target, lalu klik menu Ekstensi > Apps Script.

2. Tempel Kode

Hapus seluruh isi di Code.gs dan tempelkan kode Apps Script di bawah ini.

3. Deploy Web App

Klik Deploy > New Deployment, set akses ke "Anyone" (Siapa Saja).

// =========================================================================
// GOOGLE APPS SCRIPT BACKEND - SCHOOL PRIME PRESENSI OFFLLINE SYNC ENGINE
// target Sheet Name: "Data Absensi"
// Target GID/ID: 1-S1Xwha8cuZvNXpkXpU_3m9dam7Nu9cPU6vUYyXN5Dk
// =========================================================================

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var sheetName = "Data Absensi";
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    
    // Jika sheet belum ada, buat baru dan tambahkan header
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow([
        "NIS", 
        "Nama Lengkap", 
        "Kelas", 
        "Tanggal Absen", 
        "Keterangan", 
        "kode RFID", 
        "Waktu", 
        "Jenis Absen"
      ]);
    }
    
    var data = JSON.parse(e.postData.contents);
    
    if (Array.isArray(data)) {
      data.forEach(function(item) {
        sheet.appendRow([
          item.nis || "",
          item.nama || "",
          item.kelas || "",
          item.tanggal || "",
          item.keterangan || "Hadir",
          item.kodeRfid || "", // Kosongkan sesuai spesifikasi
          item.waktu || "",
          item.jenisAbsen || "Absen Masuk"
        ]);
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success", 
      message: "Data presensi berhasil disimpan ke sheet Data Absensi"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", 
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  try {
    var sheetName = "Data Absensi";
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: []
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var rows = sheet.getDataRange().getValues();
    var result = [];
    
    // Skip row 0 (Header)
    for (var i = 1; i < rows.length; i++) {
      result.push({
        nis: rows[i][0],
        nama: rows[i][1],
        kelas: rows[i][2],
        tanggal: rows[i][3],
        keterangan: rows[i][4],
        kodeRfid: rows[i][5],
        waktu: rows[i][6],
        jenisAbsen: rows[i][7]
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      data: result
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
